/* =====================================================
   ⚠️ 1) USERS — الأولوية القصوى، دير هادي دابا
   =====================================================

   المشكل:
     allow update: if request.auth.uid == userId;
   كانت كتخلي المستخدم يكتب أي حقل، بما فيه
   subscriptionActive. يعني جملة وحدة فالـ Console
   وكيولي مشترك بلا خلاص.

   الحل: نمنعو حقول الاشتراك من الكتابة من الواجهة.
   الخلاص كيتفعّل غير من Firebase Console (ولا Cloud
   Function بـ Admin SDK — القواعد ما كتطبقش عليه).

   ⚠️ القراءة خليناها مفتوحة للمسجّلين حيت chat.html
      كيقرا users/{uid} ديال الأصدقاء (الأسماء والصور).
      إلا حصرناها، الشات كيتخربق. شوف القسم 4.
   ===================================================== */

match /users/{userId} {

  allow read: if request.auth != null;

  allow create: if request.auth != null
    && request.auth.uid == userId
    && !request.resource.data.keys().hasAny([
         "subscriptionActive", "subscriptionEnd", "role", "isAdmin"
       ]);

  allow update: if request.auth != null
    && request.auth.uid == userId
    && !request.resource.data.diff(resource.data).affectedKeys().hasAny([
         "subscriptionActive", "subscriptionEnd", "role", "isAdmin"
       ]);

  allow delete: if false;
}

   2) GROUP INVITES

   ===================================================== */

match /groupInvites/{inviteId} {

  allow read: if request.auth != null

      resource.data.fromUid == request.auth.uid
      || resource.data.toUid == request.auth.uid
    );

}
